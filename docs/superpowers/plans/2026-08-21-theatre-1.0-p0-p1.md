```ts
import {defer} from '@dreamerbird/utils/defer'
import type {BackStorageAdapter, FullSnapshot, $IntentionalAny} from '../types'

export class BackStorage {
  private _readyD = defer<void>()
  private _adapter: BackStorageAdapter
  private _loadedState: {
    snapshot: FullSnapshot<$IntentionalAny> | null
    clock: number | null
  } | null = null

  constructor(opts: {storageAdapter: BackStorageAdapter}) {
    this._adapter = opts.storageAdapter
    void this._load()
  }

  private async _load() {
    const [snapshot, clock] = await Promise.all([
      this._adapter.get<FullSnapshot<$IntentionalAny>>('snapshot'),
      this._adapter.get<number>('clock'),
    ])
    this._loadedState = {
      snapshot: snapshot ?? null,
      clock: typeof clock === 'number' ? clock : null,
    }
    this._readyD.resolve()
  }

  get ready() {
    return this._readyD.promise
  }

  get loadedState() {
    return this._loadedState
  }

  async save(
    snapshot: FullSnapshot<$IntentionalAny>,
    clock: number | null,
  ): Promise<void> {
    await this._adapter.set('snapshot', snapshot)
    await this._adapter.set('clock', clock)
  }
}
```

```ts
// packages/saaz/src/back/BackIDBAdapter.ts
import {openDB} from 'idb'
import type {IDBPDatabase} from 'idb'
import type {BackStorageAdapter} from '../types'

export class BackIDBAdapter implements BackStorageAdapter {
  private _dbPromise: Promise<IDBPDatabase>

  constructor(idbName: string | Promise<string>) {
    this._dbPromise = Promise.resolve(idbName).then((name) =>
      openDB(name, 1, {
        upgrade(db) {
          db.createObjectStore('keyval')
        },
      }),
    )
  }

  get ready() {
    return this._dbPromise.then(() => undefined)
  }

  async get<T>(key: string): Promise<T | void> {
    const db = await this._dbPromise
    return (await db.get('keyval', key)) as T | void
  }

  async set<T>(key: string, value: T): Promise<void> {
    const db = await this._dbPromise
    await db.put('keyval', value, key)
  }
}
```

```ts
// packages/saaz/src/deferredBackend.ts
import type {SaazBackInterface} from './types'

/**
 * Wraps a backend that is chosen asynchronously (e.g. local vs tRPC depending
 * on init opts) so callers can hold a SaazBackInterface synchronously.
 */
export function createDeferredBackend(
  impl: Promise<SaazBackInterface>,
): SaazBackInterface {
  return {
    getUpdatesSinceClock: (opts) => impl.then((b) => b.getUpdatesSinceClock(opts)),
    getLastIncorporatedPeerClock: (opts) =>
      impl.then((b) => b.getLastIncorporatedPeerClock(opts)),
    closePeer: (opts) => impl.then((b) => b.closePeer(opts)),
    applyUpdates: (opts) => impl.then((b) => b.applyUpdates(opts)),
    updatePresence: (opts) => impl.then((b) => b.updatePresence(opts)),
    subscribe: (opts, onUpdate) =>
      impl.then((b) => b.subscribe(opts, onUpdate)).then((unsub) => {
        let inner: (() => void) | undefined = unsub
        return () => inner?.()
      }),
  }
}
```

```ts
/** Deletes the locally persisted saaz state for the given persistence key. */
```
